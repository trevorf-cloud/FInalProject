from openpyxl import load_workbook
from datetime import datetime, timedelta
import serial
import time

# Open serial connection to ESP32
ser = serial.Serial('COM3', 115200)

# Give ESP32 time to initialize
time.sleep(2)

# Load Excel file
FILE_PATH = r"D:\School\Scripting and Automation\FinalProject\assignments.xlsx"

wb = load_workbook(FILE_PATH)
sheet = wb.active

# Current time
now = datetime.now()

# Time window (24 hours)
alert_window = timedelta(hours=24)

alert_found = False

for row in sheet.iter_rows(min_row=2, values_only=False):

    assignment_cell = row[0]
    deadline_cell = row[1]
    status_cell = row[2]

    assignment = assignment_cell.value
    deadline = deadline_cell.value
    status = status_cell.value

    if deadline and status != "Completed":

        time_left = deadline - now

        if timedelta(0) <= time_left <= alert_window:

            print(f"ALERT: {assignment} due soon!")

            ser.write(b'LED_ON\n')

            alert_found = True
            break

if not alert_found:
    ser.write(b'LED_OFF\n')

ser.close()
